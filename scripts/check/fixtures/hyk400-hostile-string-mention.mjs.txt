// HYK-400 2R 적대 표본 ⓔ -- "--harness-dir"라는 문자열은 usage/주석에
// 언급하지만, 파서는 그 플래그를 알지도 못하고 조용히 무시한다(모르는
// 플래그를 에러로 보지 않고 그냥 건너뛴다 -- unrecognized flag 에러조차
// 안 낸다). 1R의 정적 텍스트 검사였다면 이 파일을 "지원함"으로 오탐했을
// 표본이다.
// usage: dispatch-receipt-cli.mjs --role <r> --task-label <l>
//   [--receipt-path <p>] [--harness-dir <d>]  (문서화만 됐을 뿐 처리 안 함)
export function parseDispatchReceiptArgs(args) {
  const parsed = {};
  for (let i = 0; i < args.length; i++) {
    if (args[i] === "--role") parsed.role = args[++i];
    else if (args[i] === "--task-label") parsed.harnessTaskLabel = args[++i];
    else if (args[i] === "--receipt-path") parsed.receiptPath = args[++i];
    // 그 밖의 모든 플래그(--harness-dir 포함)는 조용히 건너뛴다.
  }
  return { ok: true, ...parsed };
}
